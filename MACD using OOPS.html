  2. SELL when MACD line crosses below Signal line

class BacktestingMACD:

    def __init__(self, ticker , start , end ):
        self.ticker = ticker
        self.start = start
        self.end = end
        self.fetch_data()
        self.indicator()
        self.signal()
        self.position()
        self.returns()


    def fetch_data(self):
        self.data = yf.download(self.ticker , self.start , self.end)
        self.data = self.data.droplevel(1 , axis = 1)


    def indicator(self):
        ## MACD line creation
        self.data["ema12"] = self.data["Close"].ewm(span = 12).mean()
        self.data["ema26"] = self.data["Close"].ewm(span = 26).mean()
        self.data["MACD_line"] = self.data["ema12"] - self.data["ema26"]

        ## Signal Line creation
        self.data["signal_line"] = self.data["MACD_line"].ewm(span = 9).mean()

    def signal(self):  ## here we are generating signals 

        self.data["signal"] = np.where((self.data["MACD_line"] > self.data["signal_line"])&
                                       (self.data["MACD_line"].shift(1) < self.data["signal_line"].shift(1)), 1 , 0)

        self.data["signal"] = np.where((self.data["MACD_line"] < self.data["signal_line"])&
                                       (self.data["MACD_line"].shift(1) > self.data["signal_line"].shift(1)), -1 , self.data["signal"])

    def position(self):

        self.data["position"] = self.data["signal"].replace(to_replace=0 , method = "ffill")

    def returns(self):

        self.data["bnh"] = np.log(self.data["Close"] / self.data["Close"].shift(1))

        self.data["strategy_returns"] = self.data["position"].shift(1) * self.data["bnh"]
        print("The strategy returns:" , np.round(self.data["strategy_returns"].cumsum()[-1], 2))
        return self.data["strategy_returns"].cumsum()[-1]

    def analysis(self):

        self.data[["MACD_line" , "signal_line", "position"]].plot(figsize = (15,9) , secondary_y = "position" ,grid = True)
        self.data[["bnh" , "strategy_returns"]].cumsum().plot(figsize = (15,9) , grid = True)

        qs.reports.basic(self.data["strategy_returns"])
